import pandas as pd
import numpy as np

# Create data directly
df = pd.DataFrame({
    "name": ["Asha", "Ravi", "Meera", "Karan", "Neha"],
    "dept": ["HR", "IT", "IT", "Sales", "IT"],
    "salary": [50000, 70000, 65000, None, 72000],
    "age": [28, 34, 30, 41, 37],
    "join_date": ["2020-01-15", "2019-06-01", "2021-03-10", "2018-11-20", "2022-07-05"],
})

# Or from a list of tuples
df2 = pd.DataFrame(
    [("A", 10), ("B", 20), ("C", 30)],
    columns=["key", "value"],
)

# Quick checks
print(df.head())
print(df.info())
print(df.describe())

# Filter and select
it_staff = df[df["dept"] == "IT"][["name", "salary"]]

# Handle missing values
df["salary"] = df["salary"].fillna(df["salary"].mean())

# New columns
df["join_date"] = pd.to_datetime(df["join_date"])
df["join_year"] = df["join_date"].dt.year
df["level"] = np.where(df["age"] >= 35, "Senior", "Junior")

# Group and aggregate
summary = df.groupby("dept").agg(
    avg_salary=("salary", "mean"),
    headcount=("name", "count"),
).reset_index()

# Merge with another in-memory table
heads = pd.DataFrame({"dept": ["HR", "IT", "Sales"], "head": ["Anil", "Sunita", "Vijay"]})
merged = df.merge(heads, on="dept", how="left")

# Rank within group
df["salary_rank"] = df.groupby("dept")["salary"].rank(ascending=False)

print(summary)
